<head>
  <meta charset="UTF-8" />
  <meta name="viewport" content="width=device-width, initial-scale=1.0, maximum-scale=1.0, user-scalable=no, viewport-fit=cover" />
  <title>English Tutor</title>
  
  <!-- Telegram WebApp SDK -->
  <script src="https://telegram.org/js/telegram-web-app.js"></script>
  
  <!-- Tailwind CSS -->
  <script src="https://cdn.tailwindcss.com"></script>
  <script>
    tailwind.config = {
      theme: {
        extend: {
          colors: {
            brand: {
              bg: '#111111',
              card: '#1C1C1E',
              text: '#FFFFFF',
              hint: '#8E8E93',
              accent: '#00E5FF',
              success: '#34C759',
              warning: '#FF9500',
              danger: '#FF3B30',
            }
          },
          fontFamily: { sans: ['Inter', 'sans-serif'] }
        }
      }
    }
  </script>
  
  <!-- React & Babel (для работы React прямо в браузере) -->
  <script crossorigin src="https://unpkg.com/react@18/umd/react.production.min.js"></script>
  <script crossorigin src="https://unpkg.com/react-dom@18/umd/react-dom.production.min.js"></script>
  <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>
  
  <!-- Иконки Lucide -->
  <script src="https://unpkg.com/lucide@latest"></script>

  <!-- Шрифт и базовые стили -->
  <link href="https://fonts.googleapis.com/css2?family=Inter:wght@400;500;600;700&display=swap" rel="stylesheet">
  <style>
    body {
      background-color: #111111;
      color: #FFFFFF;
      font-family: 'Inter', sans-serif;
      margin: 0;
      padding: 0;
      overflow-x: hidden;
      -webkit-tap-highlight-color: transparent;
    }
    /* Анимации */
    @keyframes flipIn {
      0% { transform: rotateY(90deg); opacity: 0; }
      100% { transform: rotateY(0deg); opacity: 1; }
    }
    .animate-flip {
      animation: flipIn 0.4s ease-out forwards;
      transform-style: preserve-3d;
    }
    @keyframes fadeIn {
      0% { opacity: 0; transform: translateY(10px); }
      100% { opacity: 1; transform: translateY(0); }
    }
    .animate-fade-in {
      animation: fadeIn 0.3s ease-out forwards;
    }
    .safe-bottom { padding-bottom: env(safe-area-inset-bottom, 20px); }
  </style>
</head>
<body>
  <div id="root"></div>

  <script type="text/babel">
    const { useState, useEffect } = React;

    // --- ДАННЫЕ ДЛЯ ТЕСТА ---
    const INITIAL_CARDS = [
      { id: 1, en: "Serendipity", ru: "Счастливая случайность", example: "Meeting her was pure serendipity.", type: "Vocabulary" },
      { id: 2, en: "To pull strings", ru: "Потянуть за ниточки (использовать связи)", example: "He pulled some strings to get the job.", type: "Phrases" },
      { id: 3, en: "Break the ice", ru: "Растопить лед (начать общение)", example: "He told a joke to break the ice.", type: "Everyday English" },
      { id: 4, en: "Resilient", ru: "Устойчивый, стойкий", example: "She is a resilient person.", type: "Vocabulary" },
      { id: 5, en: "Bite the bullet", ru: "Стиснуть зубы (сделать что-то неприятное)", example: "I decided to bite the bullet and go to the dentist.", type: "Phrases" },
    ];

    // --- КОМПОНЕНТЫ ---
    
    // Главный экран
    const Dashboard = () => (
      <div className="p-6 animate-fade-in">
        <h1 className="text-3xl font-bold mb-2">Привет! 👋</h1>
        <p className="text-brand-hint mb-6">Готов покорять английский сегодня?</p>
        
        <div className="bg-brand-card rounded-3xl p-6 mb-4 shadow-lg border border-white/5">
          <div className="flex justify-between items-center mb-4">
            <span className="font-semibold text-lg">Твой прогресс</span>
            <span className="text-brand-accent font-bold">62%</span>
          </div>
          <div className="w-full bg-black/50 rounded-full h-3 mb-4">
            <div className="bg-brand-accent h-3 rounded-full" style={{ width: '62%' }}></div>
          </div>
          <div className="flex justify-between text-sm text-brand-hint">
            <span>Уровень: B1</span>
            <span>XP: 120 / 200</span>
          </div>
        </div>

        <button 
          onClick={() => window.location.hash = '#learn'}
          className="w-full bg-brand-accent text-black font-bold text-lg py-4 rounded-2xl shadow-[0_0_20px_rgba(0,229,255,0.3)] hover:scale-[0.98] transition-transform"
        >
          Начать обучение
        </button>
      </div>
    );

    // Экран карточек (Learn)
    const Learn = () => {
      const [currentIndex, setCurrentIndex] = useState(0);
      const [isFlipped, setIsFlipped] = useState(false);

      const card = INITIAL_CARDS[currentIndex];

      const handleNext = () => {
        setIsFlipped(false);
        setTimeout(() => {
          setCurrentIndex((prev) => (prev + 1) % INITIAL_CARDS.length);
        }, 200);
      };

      if (!card) return <div className="p-6 text-center">Загрузка...</div>;

      return (
        <div className="p-6 flex flex-col h-[85vh] animate-fade-in">
          <div className="flex justify-between items-center mb-6">
            <span className="text-brand-hint font-medium">{currentIndex + 1} / {INITIAL_CARDS.length}</span>
            <span className="bg-white/10 px-3 py-1 rounded-full text-xs">{card.type}</span>
          </div>

          {/* Карточка */}
          <div 
            onClick={() => setIsFlipped(!isFlipped)}
            className={`flex-1 bg-brand-card rounded-3xl p-8 flex flex-col items-center justify-center cursor-pointer shadow-2xl border border-white/10 transition-transform ${isFlipped ? 'bg-brand-card/80' : ''}`}
          >
            {!isFlipped ? (
              <div className="text-center animate-fade-in">
                <h2 className="text-3xl font-bold mb-4 text-white">{card.en}</h2>
                <p className="text-brand-hint text-sm">Нажми, чтобы увидеть перевод</p>
              </div>
            ) : (
              <div className="text-center animate-fade-in">
                <h2 className="text-2xl font-bold mb-2 text-brand-accent">{card.ru}</h2>
                <div className="mt-6 p-4 bg-black/30 rounded-xl">
                  <p className="text-sm italic text-gray-300">"{card.example}"</p>
                </div>
              </div>
            )}
          </div>

          {/* Кнопки оценки (появляются только после переворота) */}
          <div className={`grid grid-cols-4 gap-2 mt-6 transition-opacity duration-300 ${isFlipped ? 'opacity-100' : 'opacity-0 pointer-events-none'}`}>
            <button onClick={handleNext} className="bg-brand-danger text-white py-3 rounded-xl font-semibold text-sm">Again</button>
            <button onClick={handleNext} className="bg-brand-warning text-black py-3 rounded-xl font-semibold text-sm">Hard</button>
            <button onClick={handleNext} className="bg-brand-success text-black py-3 rounded-xl font-semibold text-sm">Good</button>
            <button onClick={handleNext} className="bg-blue-500 text-white py-3 rounded-xl font-semibold text-sm">Easy</button>
          </div>
          
          {!isFlipped && <p className="text-center text-brand-hint mt-6 text-sm h-10">Кнопки появятся после того, как перевернешь карточку</p>}
          {isFlipped && <div className="h-10 mt-6"></div>}
        </div>
      );
    };

    // Заглушки для остальных экранов
    const Cards = () => <div className="p-6 animate-fade-in"><h1 className="text-2xl font-bold mb-4">Мои карточки</h1><p className="text-brand-hint">Здесь будет список всех слов.</p></div>;
    const Progress = () => <div className="p-6 animate-fade-in"><h1 className="text-2xl font-bold mb-4">Статистика</h1><p className="text-brand-hint">Здесь будут графики.</p></div>;
    const Profile = () => <div className="p-6 animate-fade-in"><h1 className="text-2xl font-bold mb-4">Профиль</h1><p className="text-brand-hint">Здесь будут настройки.</p></div>;

    // Нижняя навигация
    const BottomNav = ({ currentPath }) => {
      const navItems = [
        { path: '#dashboard', icon: 'home', label: 'Главная' },
        { path: '#learn', icon: 'book-open', label: 'Учеба' },
        { path: '#cards', icon: 'layers', label: 'Карточки' },
        { path: '#progress', icon: 'bar-chart-2', label: 'Прогресс' },
        { path: '#profile', icon: 'user', label: 'Профиль' },
      ];

      useEffect(() => {
        if (window.lucide) window.lucide.createIcons();
      }, [currentPath]);

      return (
        <div className="fixed bottom-0 left-0 right-0 bg-[#1a1a1c] border-t border-white/10 px-2 py-2 flex justify-around items-center z-50 safe-bottom">
          {navItems.map((item) => {
            const isActive = currentPath === item.path;
            return (
              <a 
                key={item.path} 
                href={item.path}
                className={`flex flex-col items-center p-2 rounded-xl transition-colors w-16 ${isActive ? 'text-brand-accent' : 'text-brand-hint'}`}
              >
                <i data-lucide={item.icon} className="w-6 h-6"></i>
                <span className="text-[10px] mt-1 font-medium">{item.label}</span>
              </a>
            );
          })}
        </div>
      );
    };

    // --- ГЛАВНЫЙ КОМПОНЕНТ ---
    const App = () => {
      const [route, setRoute] = useState(window.location.hash || '#dashboard');

      useEffect(() => {
        // Инициализация Telegram
        if (window.Telegram?.WebApp) {
          const tg = window.Telegram.WebApp;
          tg.ready();
          tg.expand();
          tg.setHeaderColor('#111111');
        }

        // Обработка навигации
        const handleHashChange = () => setRoute(window.location.hash || '#dashboard');
        window.addEventListener('hashchange', handleHashChange);
        return () => window.removeEventListener('hashchange', handleHashChange);
      }, []);

      const renderScreen = () => {
        switch(route) {
          case '#learn': return <Learn />;
          case '#cards': return <Cards />;
          case '#progress': return <Progress />;
          case '#profile': return <Profile />;
          default: return <Dashboard />;
        }
      };

      return (
        <div className="min-h-screen pb-24">
          {renderScreen()}
          <BottomNav currentPath={route} />
        </div>
      );
    };

    const root = ReactDOM.createRoot(document.getElementById('root'));
    root.render(<App />);
  </script>
</body>
</html>
